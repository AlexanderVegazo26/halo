# HALO Vulkan/RADV backend (TRD §3.2, §20, §41).
#
# Shaders under shaders/ are compiled to SPIR-V by glslc at build time, validated with
# spirv-val when available, and embedded into the library together with the SHA-256 of
# each SPIR-V blob (for pipeline-cache validation). A missing glslc fails configure.

find_package(Vulkan REQUIRED)

find_program(HALO_GLSLC glslc)
if(NOT HALO_GLSLC)
  message(FATAL_ERROR "HALO_BUILD_VULKAN=ON requires glslc (shaderc) to compile compute "
                      "shaders (TRD §41). Install shaderc/glslc or configure with "
                      "-DHALO_BUILD_VULKAN=OFF.")
endif()
find_program(HALO_SPIRV_VAL spirv-val)
if(NOT HALO_SPIRV_VAL)
  message(WARNING "spirv-val not found: embedded SPIR-V will not be validated at build time")
endif()

set(HALO_VK_TARGET_ENV vulkan1.3)

# name  ->  shaders/<path>.comp   (name must be unique; it is the lookup key at runtime)
set(HALO_VK_SHADERS
  norm/rms_norm
  matmul/matvec_f32
  matmul/matvec_q8_0
  matmul/matvec_q4_k
  matmul/matvec_q5_k
  matmul/matvec_q6_k
  matmul/matvec_iq4_xs
  matmul/matvec_iq4_nl
  matmul/matvec_q3_k
  matmul/matvec_iq3_s
  linear_attn/gated_delta_rule_decode
  linear_attn/gated_delta_rule_chunked
  linear_attn/gdn_gcheck
  linear_attn/conv1d_silu
  norm/gated_rms_norm
  norm/add_rms_norm
  eltwise/eltwise
  eltwise/gdn_gates
  rope/rope_neox
  get_rows/get_rows_f32
  get_rows/get_rows_q8_0
  get_rows/get_rows_q4_k
  get_rows/get_rows_q5_k
  get_rows/get_rows_q6_k
  get_rows/get_rows_iq4_xs
  get_rows/get_rows_iq4_nl
  get_rows/get_rows_q3_k
  get_rows/get_rows_iq3_s
  sample/topk
  attention/kv_write
  attention/attention
  reduce/argmax_partial
  reduce/argmax_final
)

set(_shader_dir ${CMAKE_CURRENT_SOURCE_DIR}/shaders)
set(_gen_dir ${CMAKE_CURRENT_BINARY_DIR}/generated)
file(MAKE_DIRECTORY ${_gen_dir} ${CMAKE_CURRENT_BINARY_DIR}/spirv)

set(_gen_sources)
set(_decls "")
set(_entries "")
foreach(rel IN LISTS HALO_VK_SHADERS)
  get_filename_component(_name ${rel} NAME)
  set(_src ${_shader_dir}/${rel}.comp)
  set(_spv ${CMAKE_CURRENT_BINARY_DIR}/spirv/${_name}.spv)
  set(_cpp ${_gen_dir}/spv_${_name}.cpp)
  set(_id k_shader_${_name})

  set(_val_cmd)
  if(HALO_SPIRV_VAL)
    set(_val_cmd COMMAND ${HALO_SPIRV_VAL} --target-env ${HALO_VK_TARGET_ENV} ${_spv})
  endif()
  # No -g: debug info would embed source paths and break reproducibility.
  add_custom_command(
    OUTPUT ${_spv}
    COMMAND ${HALO_GLSLC} --target-env=${HALO_VK_TARGET_ENV} -O -Werror
            -I ${_shader_dir}/common -MD -MF ${_spv}.d -o ${_spv} ${_src}
    ${_val_cmd}
    DEPENDS ${_src}
    DEPFILE ${_spv}.d
    COMMENT "glslc ${rel}.comp"
    VERBATIM)
  add_custom_command(
    OUTPUT ${_cpp}
    COMMAND ${CMAKE_COMMAND} -DSPV=${_spv} -DID=${_id} -DNAME=${_name}
            -DSOURCE=${rel}.comp -DOUT=${_cpp}
            -P ${CMAKE_CURRENT_SOURCE_DIR}/cmake/embed_spirv.cmake
    DEPENDS ${_spv} ${CMAKE_CURRENT_SOURCE_DIR}/cmake/embed_spirv.cmake
    COMMENT "embed ${_name}.spv"
    VERBATIM)
  list(APPEND _gen_sources ${_cpp})
  string(APPEND _decls "extern constinit const EmbeddedShader ${_id};\n")
  string(APPEND _entries "    &${_id},\n")
endforeach()

# The registry table is known at configure time (names only; bytes and hashes come from
# the per-shader generated sources above).
file(CONFIGURE OUTPUT ${_gen_dir}/shader_table.cpp CONTENT
"// GENERATED by backends/vulkan/CMakeLists.txt — do not edit.
#include \"shader_registry.h\"

namespace halo::vulkan::detail {
${_decls}
constinit const EmbeddedShader* const k_shader_table[] = {
${_entries}};
constinit const std::span<const EmbeddedShader* const> k_shaders{k_shader_table};
}  // namespace halo::vulkan::detail
" @ONLY)

add_library(halo_backend_vulkan STATIC
  src/context.cpp
  src/device_info.cpp
  src/buffer.cpp
  src/kernel.cpp
  src/stream.cpp
  src/ops.cpp
  src/ops_layer.cpp
  src/ops_paged.cpp
  src/ops_head.cpp
  src/shaders.cpp
  src/sha256.cpp
  src/vk_util.cpp
  ${_gen_sources}
  ${_gen_dir}/shader_table.cpp)
target_include_directories(halo_backend_vulkan
  PUBLIC ${PROJECT_SOURCE_DIR}/include
  PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/src)
# halo_tensor is PUBLIC: the public ops.h uses halo::DType (halo/tensor/dtype.h), code
# review N-2.
if(NOT TARGET halo_tensor)
  message(FATAL_ERROR "backends/vulkan needs halo_tensor: add 'tensor' to HALO_ONLY")
endif()
target_link_libraries(halo_backend_vulkan PUBLIC halo_core halo_tensor Vulkan::Vulkan PRIVATE halo_warnings)
