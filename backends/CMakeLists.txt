foreach(be IN ITEMS cpu vulkan hip)
  halo_selected(${be} _sel)
  if(be STREQUAL "vulkan" AND NOT HALO_BUILD_VULKAN)
    set(_sel FALSE)
  endif()
  if(be STREQUAL "hip" AND NOT HALO_BUILD_HIP)
    set(_sel FALSE)
  endif()
  if(_sel AND EXISTS ${CMAKE_CURRENT_SOURCE_DIR}/${be}/CMakeLists.txt)
    add_subdirectory(${be})
  endif()
endforeach()
